### 9.1 数据模型设计

**文件**：`server/prisma/schema.prisma`

```prisma
model Information {
  id           String   @id @default(cuid()) @db.VarChar
  title        String   @db.VarChar          // 标题
  content      String   @db.Text             // 内容
  category     String   @default("other") @db.VarChar  // business, supply, notice, other
  contactname  String?  @db.VarChar          // 联系人
  contactphone String?  @db.VarChar          // 联系电话
  publisherid  String   @db.VarChar          // 发布者 User.id
  createdat    DateTime @default(now) @db.Timestamp(6)
  updatedat    DateTime @default(now) @db.Timestamp(6)
  
  @@index([category])
  @@index([publisherid])
  @@index([createdat])
}
```

**字段命名说明**：采用 snake_case 风格（`createdat`、`updatedat`、`publisherid`），与现有 `Activity`、`Member` 等模型保持一致。API 响应层需进行字段转换（参考 `transformActivity` 函数）。

### 9.2.1 异常处理定义

**错误码规范**：

| HTTP 状态码 | 错误码 | 触发场景 | 响应示例 |
|------------|--------|----------|----------|
| 400 | `INVALID_INPUT` | 请求参数校验失败（标题/内容为空、分类无效、手机号格式错误） | `{ success: false, error: '标题不能为空', code: 'INVALID_INPUT' }` |
| 401 | - | 未登录或 Token 无效 | `{ success: false, error: '请先登录' }` |
| 403 | `UNAUTHORIZED_DELETE` | 非管理员且非发布者尝试删除信息 | `{ success: false, error: '无删除权限', code: 'UNAUTHORIZED_DELETE' }` |
| 403 | `UNAUTHORIZED_UPDATE` | 非发布者本人尝试更新信息 | `{ success: false, error: '仅发布者可编辑', code: 'UNAUTHORIZED_UPDATE' }` |
| 404 | `INFO_NOT_FOUND` | 信息 ID 不存在 | `{ success: false, error: '信息不存在', code: 'INFO_NOT_FOUND' }` |
| 500 | `SERVER_ERROR` | 服务器内部错误（数据库异常、未捕获错误） | `{ success: false, error: '服务器内部错误', code: 'SERVER_ERROR' }` |

### 9.3 Store 层设计

**文件**：`server/src/models/Information.ts`（新建）

```typescript
import { PrismaClient, Information as InformationType } from '@prisma/client';

const prisma = new PrismaClient();

export interface InfoFilter {
  page: number;
  limit: number;
  category?: string;
  publisherId?: string;
}

export class InformationStore {
  async getPaginated(filter: InfoFilter): Promise<{ data: InformationType[]; total: number }> {
    const { page, limit, category, publisherId } = filter;
    
    const where: any = {};
    if (category && category !== 'all') {
      where.category = category;
    }
    if (publisherId) {
      where.publisherId = publisherId;
    }
    
    const [data, total] = await Promise.all([
      prisma.information.findMany({
        where,
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { createdAt: 'desc' },
      }),
      prisma.information.count({ where }),
    ]);
    
    return { data, total };
  }
  
  async getById(id: string): Promise<InformationType | null> {
    return prisma.information.findUnique({ where: { id } });
  }
  
  async create(data: Omit<InformationType, 'id' | 'createdAt' | 'updatedAt'>): Promise<InformationType> {
    return prisma.information.create({ data });
  }
  
  async delete(id: string): Promise<void> {
    await prisma.information.delete({ where: { id } });
  }
  
  async update(id: string, data: Partial<InformationType>): Promise<InformationType> {
    return prisma.information.update({ where: { id }, data });
  }
  
  async getCategories(): Promise<string[]> {
    const result = await prisma.information.groupBy({
      by: ['category'],
    });
    return result.map(r => r.category);
  }
}

export const informationStore = new InformationStore();
```

### 9.5 权限中间件设计（🆕 优化项）

**文件**：`server/src/middleware/info-auth.ts`（新建）

```typescript
import { Request, Response, NextFunction } from 'express';
import { informationStore } from '../models/Information';

/**
 * 授权中间件 - 检查用户是否有权删除该信息
 * 规则：管理员 或 发布者本人
 */
export async function authorizeInfoDelete(
  req: Request, 
  res: Response, 
  next: NextFunction
): Promise<void> {
  try {
    const info = await informationStore.getById(req.params.id);
    
    if (!info) {
      res.status(404).json({ success: false, error: '信息不存在' });
      return;
    }
    
    const user = req.user;
    if (!user) {
      res.status(401).json({ success: false, error: '未认证' });
      return;
    }
    
    const isAdmin = (process.env.ADMIN_PHONES?.split(',').map(s => s.trim()).includes(user.phone));
    const isOwner = info.publisherId === user.id;
    
    if (!isAdmin && !isOwner) {
      res.status(403).json({ success: false, error: '无删除权限' });
      return;
    }
    
    next();
  } catch (error) {
    console.error('Authorize info delete error:', error);
    res.status(500).json({ success: false, error: '权限检查失败' });
  }
}

/**
 * 授权中间件 - 检查用户是否有权更新该信息
 * 规则：仅发布者本人
 */
export async function authorizeInfoUpdate(
  req: Request, 
  res: Response, 
  next: NextFunction
): Promise<void> {
  try {
    const info = await informationStore.getById(req.params.id);
    
    if (!info) {
      res.status(404).json({ success: false, error: '信息不存在' });
      return;
    }
    
    const isOwner = info.publisherId === req.user?.id;
    
    if (!isOwner) {
      res.status(403).json({ success: false, error: '仅发布者可编辑' });
      return;
    }
    
    next();
  } catch (error) {
    console.error('Authorize info update error:', error);
    res.status(500).json({ success: false, error: '权限检查失败' });
  }
}
```
